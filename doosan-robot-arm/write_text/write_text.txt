def tw_fail(msg):
    """오류를 TP 알람 팝업으로 알리고 프로그램을 멈춥니다."""
    tp_popup("[write_text] " + msg, DR_PM_ALARM, 1)
    exit()
    raise RuntimeError(msg)  # exit() 가 즉시 멈추지 않는 경우에도 모션이 나가지 않도록


def tw_dist(a, b):
    return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5


def tw_check_settings():
    """TEXT 와 숫자 설정값을 검사합니다."""
    if not isinstance(TEXT, str) or len(TEXT) == 0:
        tw_fail("TEXT is empty.")
    if len(TEXT) > TW_MAX_TEXT_LEN:
        tw_fail("TEXT has {0} chars (max {1}).".format(len(TEXT), TW_MAX_TEXT_LEN))
    bad = []
    for ch in TEXT:
        if ch not in TW_FONT:
            label = ch if ord(ch) < 128 else "U+%04X" % ord(ch)
            if label not in bad:
                bad.append(label)
    if bad:
        tw_fail("Unsupported characters in TEXT: " + " ".join(bad))

    for name, value, lo, hi in (
        ("PEN_UP_Z", PEN_UP_Z, 1.0, 200.0),
        ("CHAR_HEIGHT", CHAR_HEIGHT, 0.0, 1000.0),
        ("CANVAS_MARGIN", CANVAS_MARGIN, 0.0, 500.0),
        ("LETTER_SPACING", LETTER_SPACING, 0.0, 2.0),
        ("DRY_RUN_OFFSET_Z", DRY_RUN_OFFSET_Z, 0.0, 200.0),
        ("VEL_DRAW", VEL_DRAW, 1.0, 1000.0),
        ("VEL_TRAVEL", VEL_TRAVEL, 1.0, 1000.0),
        ("VEL_PEN", VEL_PEN, 1.0, 500.0),
        ("ACC", ACC, 10.0, 5000.0),
        ("BLEND_RADIUS", BLEND_RADIUS, 0.0, 10.0),
    ):
        try:
            v = float(value)
        except Exception:
            tw_fail(name + " must be a number.")
        if v < lo or v > hi:
            tw_fail("{0}={1} is out of range [{2}, {3}].".format(name, value, lo, hi))
    if PEN_DOWN_Z is not None:
        try:
            float(PEN_DOWN_Z)
        except Exception:
            tw_fail("PEN_DOWN_Z must be None or a number.")


def tw_corners():
    """4꼭지점을 [TL, TR, BR, BL] float 좌표로 만들고 볼록 사각형/순서를 검사합니다."""
    names = ("CANVAS_TOP_LEFT", "CANVAS_TOP_RIGHT", "CANVAS_BOTTOM_RIGHT", "CANVAS_BOTTOM_LEFT")
    raw = (CANVAS_TOP_LEFT, CANVAS_TOP_RIGHT, CANVAS_BOTTOM_RIGHT, CANVAS_BOTTOM_LEFT)
    pts = []
    for name, p in zip(names, raw):
        try:
            if len(p) != 2:
                raise ValueError()
            pts.append((float(p[0]), float(p[1])))
        except Exception:
            tw_fail(name + " must be (x, y) in mm.")

    # 위에서(+z) 내려다볼 때 TL -> TR -> BR -> BL 은 시계 방향이어야 정상(외적 < 0)
    crosses = []
    for i in range(4):
        a, b, c = pts[i], pts[(i + 1) % 4], pts[(i + 2) % 4]
        crosses.append((b[0] - a[0]) * (c[1] - b[1]) - (b[1] - a[1]) * (c[0] - b[0]))
    if all(cr < 0 for cr in crosses):
        pass
    elif all(cr > 0 for cr in crosses):
        if CHECK_CORNER_ORDER:
            tw_fail("Corner order is mirrored. Swap LEFT and RIGHT corners.")
    else:
        tw_fail("Corners must form a convex quad in order TL, TR, BR, BL.")
    return pts


def tw_layout(width_mm, height_mm):
    """TEXT 를 배치하고 캔버스에 맞게 크기/위치를 정합니다.

    반환: ([(문자, [획, ...]), ...], 대문자 높이 mm)
          획의 점 (s, t) 는 캔버스 좌하단 기준 mm 좌표 (s: 오른쪽, t: 위쪽)
    """
    spacing = LETTER_SPACING * TW_CAP
    placed = []
    cursor = 0.0
    y_lo, y_hi = 0.0, TW_CAP
    for i, ch in enumerate(TEXT):
        advance, glyph = TW_FONT[ch]
        strokes = []
        for stroke in glyph:
            pts = [(cursor + p[0], float(p[1])) for p in stroke]
            for p in pts:
                y_lo = min(y_lo, p[1])
                y_hi = max(y_hi, p[1])
            strokes.append(pts)
        placed.append((ch, strokes))
        cursor += advance
        if i < len(TEXT) - 1:
            cursor += spacing
    text_w = cursor
    text_h = y_hi - y_lo

    avail_w = width_mm - 2.0 * CANVAS_MARGIN
    avail_h = height_mm - 2.0 * CANVAS_MARGIN
    if avail_w <= 0 or avail_h <= 0:
        tw_fail("CANVAS_MARGIN is too large for canvas {0:.0f} x {1:.0f} mm.".format(width_mm, height_mm))
    fit = avail_h / text_h
    if text_w > 0:
        fit = min(fit, avail_w / text_w)
    if CHAR_HEIGHT > 0:
        scale = CHAR_HEIGHT / TW_CAP
        if scale > fit * 1.0001:
            tw_fail("CHAR_HEIGHT {0:.1f} mm does not fit. Max is {1:.1f} mm.".format(CHAR_HEIGHT, fit * TW_CAP))
    else:
        scale = fit

    # 가운데 정렬
    s0 = CANVAS_MARGIN + (avail_w - text_w * scale) / 2.0
    t0 = CANVAS_MARGIN + (avail_h - text_h * scale) / 2.0 - y_lo * scale
    result = []
    for ch, strokes in placed:
        result.append((ch, [[(s0 + x * scale, t0 + y * scale) for (x, y) in st] for st in strokes]))
    return result, scale * TW_CAP


def tw_to_base(corners, width_mm, height_mm, s, t):
    """캔버스 좌표 (s, t)[mm] -> 베이스 (x, y)[mm]. 4꼭지점 쌍선형 보간."""
    tl, tr, br, bl = corners
    u = s / width_mm
    v = t / height_mm
    x = (1 - u) * (1 - v) * bl[0] + u * (1 - v) * br[0] + u * v * tr[0] + (1 - u) * v * tl[0]
    y = (1 - u) * (1 - v) * bl[1] + u * (1 - v) * br[1] + u * v * tr[1] + (1 - u) * v * tl[1]
    return (x, y)


def tw_clean(stroke):
    """연속 중복점을 제거합니다 (거리 0 모션 방지)."""
    out = []
    for p in stroke:
        if not out or tw_dist(out[-1], p) > TW_EPS:
            out.append(p)
    return out


def tw_move(x, y, z, ori, vel, r=0.0):
    movel(posx(x, y, z, ori[0], ori[1], ori[2]), vel=vel, acc=ACC, radius=r, ref=DR_BASE)


def tw_draw_stroke(pts, z_down, z_up, ori):
    """획 하나: 시작점 위로 이동 -> 펜 내림 -> 획 따라 쓰기 -> 펜 올림."""
    tw_move(pts[0][0], pts[0][1], z_up, ori, VEL_TRAVEL)
    tw_move(pts[0][0], pts[0][1], z_down, ori, VEL_PEN)
    n = len(pts)
    for k in range(1, n):
        r = 0.0
        if BLEND_RADIUS > 0 and k < n - 1:
            r = min(BLEND_RADIUS, 0.4 * tw_dist(pts[k - 1], pts[k]), 0.4 * tw_dist(pts[k], pts[k + 1]))
        tw_move(pts[k][0], pts[k][1], z_down, ori, VEL_DRAW, r)
    tw_move(pts[-1][0], pts[-1][1], z_up, ori, VEL_PEN)


def tw_main():
    tw_check_settings()
    corners = tw_corners()
    tl, tr, br, bl = corners
    width_mm = (tw_dist(tl, tr) + tw_dist(bl, br)) / 2.0
    height_mm = (tw_dist(tl, bl) + tw_dist(tr, br)) / 2.0
    if width_mm < 10.0 or height_mm < 10.0:
        tw_fail("Canvas is too small ({0:.0f} x {1:.0f} mm).".format(width_mm, height_mm))

    chars, cap_mm = tw_layout(width_mm, height_mm)
    jobs = []
    n_strokes = 0
    for ch, strokes in chars:
        xy = []
        for st in strokes:
            st_xy = tw_clean([tw_to_base(corners, width_mm, height_mm, s, t) for (s, t) in st])
            if st_xy:
                xy.append(st_xy)
        jobs.append((ch, xy))
        n_strokes += len(xy)
    if n_strokes == 0:
        tw_fail("TEXT has nothing to draw.")

    # 이전 태스크에서 세팅한 현재 위치: z = 펜 다운 높이, 자세는 그대로 사용
    cur, sol = get_current_posx(ref=DR_BASE)
    start = [float(cur[i]) for i in range(6)]
    ori = start[3:6]
    z_down = start[2] if PEN_DOWN_Z is None else float(PEN_DOWN_Z)
    z_down += DRY_RUN_OFFSET_Z
    z_up = z_down + PEN_UP_Z

    if CONFIRM_BEFORE_START:
        tp_popup("Write '{0}' / char height {1:.1f} mm / {2} strokes / pen z {3:.1f}, up +{4:.1f}. "
                 "Press Resume to start.".format(TEXT, cap_mm, n_strokes, z_down, PEN_UP_Z),
                 DR_PM_MESSAGE)

    # 시작 위치에서 수직으로 펜 들기 (이미 더 높이 있으면 생략)
    if start[2] < z_up - TW_EPS:
        tw_move(start[0], start[1], z_up, ori, VEL_PEN)

    for idx, (ch, strokes) in enumerate(jobs):
        if not strokes:
            continue
        tp_log("write_text: {0}/{1} '{2}'".format(idx + 1, len(jobs), ch))
        for st in strokes:
            tw_draw_stroke(st, z_down, z_up, ori)

    if RETURN_TO_START:
        tw_move(start[0], start[1], max(start[2], z_up), ori, VEL_TRAVEL)
    tp_log("write_text: done '{0}'".format(TEXT))


# =============================================================================
#  [4] 실행
# =============================================================================
tw_main()
